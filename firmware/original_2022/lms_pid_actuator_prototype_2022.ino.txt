// Lexus Mobile Showroom (LMS) - PID position control prototype
// Autostarke R&D, 2022
//
// Bench prototype for the closed loop control of the side extension actuators.
// Two encoder motors (M1, M2) stand in for the linear actuators.
// The motors run and print the encoder angle (0-359) until a setpoint is typed
// into the Serial Monitor. The PID loop then drives both motors to that setpoint.
//
// Tuned by trial and error: Kp = 5, Ki = 20, Kd = 10 (see README for the tuning runs).
// Transcribed from the thesis appendix.

#define MAX 12
#define MIN 0

int A_1 = 2;
int A_2 = 1;
int motorP1 = 5;
int motorN1 = 6;
int motorP2 = 8;
int motorN2 = 9;

float position = 360;
float angle = 0;
int disp = 0;
int encoder = 0;
float control = 255;
int measure_period = 1;
int sample_period = 1000;

void setup() {
  pinMode(A_1, INPUT_PULLUP);
  pinMode(A_2, INPUT_PULLUP);
  pinMode(motorP1, OUTPUT);
  pinMode(motorP2, OUTPUT);
  pinMode(motorN1, OUTPUT);
  pinMode(motorN2, OUTPUT);
  attachInterrupt(0, count, RISING);
  Serial.begin(115200);
}

void loop() {
  disp = (int) angle;
  disp %= 360;

  int out1 = motorRun(motorP1, motorN1, position, (float) disp, control);
  int out2 = motorRun(motorP2, motorN2, position, (float) disp, control);

  if (Serial.available() > 0) {
    position = Serial.parseFloat();
  }

  float control = pid_control(position, (float) disp);

  angle += (out1 * measure());
  Serial.println(disp);
  delay(100);
}

float pid_control(float setpoint, float process) {
  static float pre_error = 0;
  static float integral = 0;
  float error, derivative, output;
  float Kp = 5;
  float Ki = 20;
  float Kd = 10;

  error = setpoint - process;
  integral += error * sample_period / 1000;
  derivative = (error - pre_error) / (sample_period / 1000);
  output = Kp * error + Ki * integral + Kd * derivative;

  if (output > MAX) {
    output = MAX;
  } else if (output < MIN) {
    output = MIN;
  }
  pre_error = error;
  return output;
}

float measure() {
  encoder = 0;
  delay(measure_period);
  return encoder;
}

void count() {
  encoder++;
}

int motorRun(int m1, int m2, float setpoint, float process, float cont) {
  int output = 1;
  float diff = setpoint - process;

  if (setpoint > process) {
    analogWrite(m1, cont);
    analogWrite(m2, 0);
  } else if (setpoint < process) {
    analogWrite(m1, 0);
    analogWrite(m2, cont);
    output = -1;
  } else if (abs(diff) < 10) {
    analogWrite(m1, 0);
    analogWrite(m2, 0);
  }
  return output;
}
